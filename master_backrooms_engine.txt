    local function safeSetMaterial(ent, mat, fallbackColor)
        if not isValid(ent) then return end
        local success = false
        if mat and mat ~= "" then
            local ok, _ = pcall(ent.setMaterial, ent, mat)
            if ok then success = true end
        end
        if not success or fallbackColor then
            if not success then
                pcall(ent.setMaterial, ent, "models/debug/debugwhite")
            end
            if fallbackColor then
                pcall(ent.setColor, ent, fallbackColor)
            end
        end
    end

    local function safeSetSubMaterial(ent, idx, mat)
        if not isValid(ent) or not mat or mat == "" then return end
        pcall(ent.setSubMaterial, ent, idx, mat)
    end

    local function safeSetColor(ent, col)
        if not isValid(ent) or not col then return end
        pcall(ent.setColor, ent, col)
    end

        if not isValid(ent) or not idx then return end

        if not isValid(ent) or not bone or not ang then return end

    local function carve(cx, cy)
        grid[cx][cy].visited = true
        local directions = {
            { x = 0, y = 1, wall = "north", opp = "south" },
            { x = 0, y = -1, wall = "south", opp = "north" },
            { x = 1, y = 0, wall = "east", opp = "west" },
            { x = -1, y = 0, wall = "west", opp = "east" }
        }

        for i = #directions, 2, -1 do
            local j = math.random(i)
            directions[i], directions[j] = directions[j], directions[i]
        end

        for _, d in ipairs(directions) do
            local nx, ny = cx + d.x, cy + d.y
            if nx >= 1 and nx <= CONFIG.GRID_W and ny >= 1 and ny <= CONFIG.GRID_H then
                if not grid[nx][ny].visited then
                    grid[cx][cy][d.wall] = false
                    grid[nx][ny][d.opp] = false
                    carve(nx, ny)
                elseif math.random() < 0.15 then
                    grid[cx][cy][d.wall] = false
                    grid[nx][ny][d.opp] = false
                end
            end
        end
    end

    -- --------------------------------------------------------------------------
    -- 7. PROXIMITY-TRIGGERED VOID MATERIALIZATION SEQUENCE
    -- --------------------------------------------------------------------------
    local function triggerMaterialize(gate)
        gate.materialized = true
        print("[Backrooms Engine] ⚡ Void Threshold Activated! Materializing sector at " .. tostring(gate.pos))

        -- 1. Broadcast HL1 Xen Green Zap Sequence to all clients
        net.start("BackroomsVoidZap")
        net.writeVector(gate.pos)
        net.writeAngle(gate.ang)
        net.send()

        -- 2. Delay for 0.45s of violent crackling lightning, then dematerialize barrier & expand
        timer.simple(0.45, function()
            if isValid(gate.barrier) then
                pcall(gate.barrier.remove, gate.barrier)
                gate.barrier = nil
            end

            -- Enqueue New Corridor Extension continuing through the gateway
            local stepVec = gate.extDir * CONFIG.CELL_SIZE
            local curCenter = gate.pos - Vector(0, 0, CONFIG.WALL_HEIGHT * 0.5)

            for step = 1, 3 do
                local extCell = curCenter + (stepVec * step)
                
                -- Floor plate for new hallway
                table.insert(spawnQueue, {
                    pos = extCell,
                    ang = Angle(0, 0, 0),
                    model = "models/hunter/plates/plate4x4.mdl",
                    mat = FLOOR_PRESETS[1].mat,
                    color = FLOOR_PRESETS[1].color,
                    category = "floor"
                })

                -- Flanking walls
                local rightOffset = Angle(0, gate.ang.yaw, 0):getRight() * (CONFIG.CELL_SIZE * 0.5)
                table.insert(spawnQueue, {
                    pos = extCell + rightOffset + Vector(0, 0, CONFIG.WALL_HEIGHT * 0.5),
                    ang = Angle(90, gate.ang.yaw + 90, 0),
                    model = CONFIG.WALL_MODEL,
                    mat = WALL_PRESETS[1].mat,
                    color = WALL_PRESETS[1].color,
                    category = "wall"
                })
                table.insert(spawnQueue, {
                    pos = extCell - rightOffset + Vector(0, 0, CONFIG.WALL_HEIGHT * 0.5),
                    ang = Angle(90, gate.ang.yaw + 90, 0),
                    model = CONFIG.WALL_MODEL,
                    mat = WALL_PRESETS[1].mat,
                    color = WALL_PRESETS[1].color,
                    category = "wall"
                })

                -- Terminal Wall at end of extension
                if step == 3 then
                    table.insert(spawnQueue, {
                        pos = extCell + (gate.extDir * (CONFIG.CELL_SIZE * 0.5)) + Vector(0, 0, CONFIG.WALL_HEIGHT * 0.5),
                        ang = Angle(90, gate.ang.yaw, 0),
                        model = CONFIG.WALL_MODEL,
                        mat = WALL_PRESETS[1].mat,
                        color = WALL_PRESETS[1].color,
                        category = "wall"
                    })
                end
            end
        end)
    end

    -- Monitor player distance to active Void Gateways every 0.35s
    timer.create("VoidGatewayMonitor", 0.35, 0, function()
        local plys = getAllPlayers()
        for _, gate in ipairs(voidGateways) do
            if not gate.materialized then
                for _, ply in ipairs(plys) do
                    if isValid(ply) and ply:getPos():getDistance(gate.pos) <= CONFIG.VOID_TRIGGER_DIST then
                        triggerMaterialize(gate)
                        break
                    end
                end
            end
        end
    end)

    -- --------------------------------------------------------------------------
    -- 8. ASYNCHRONOUS BURST-REGULATED FIFO SPAWNER WITH PCALL PROTECTION
    -- --------------------------------------------------------------------------
    local spawnIndex = 1
    local spawnedCount = 0

    timer.create("BackroomsSafeSpawner", CONFIG.SPAWN_INTERVAL, 0, function()
        if spawnIndex > #spawnQueue then
            return -- Continues listening as new extensions are enqueued
        end

        if prop.canSpawn() then
            local item = spawnQueue[spawnIndex]
            local ok, p = pcall(prop.create, item.pos, item.ang, item.model, true)
            if ok and isValid(p) then
                if item.category == "wall" or item.category == "floor" or item.category == "void_barrier" then
                    safeSetMaterial(p, item.mat, item.color)
                end

                if item.onSpawn then
                    pcall(item.onSpawn, p)
                end
                
                table.insert(activeEntities, p)
                spawnedCount = spawnedCount + 1
                spawnIndex = spawnIndex + 1

                if spawnedCount % 20 == 0 or spawnedCount == #spawnQueue then
                    local pct = math.floor((spawnedCount / #spawnQueue) * 100)
                    print("[Backrooms Engine] Materializing: " .. spawnedCount .. "/" .. #spawnQueue .. " (" .. pct .. "%)")
                end
            else
                spawnIndex = spawnIndex + 1
            end
        end
    end)

    -- --------------------------------------------------------------------------
    -- 9. CLEANUP ON REMOVAL
    -- --------------------------------------------------------------------------
    hook.add("Removed", "BackroomsCleanup", function()
        timer.remove("BackroomsSafeSpawner")
        timer.remove("VoidGatewayMonitor")
        for _, ent in ipairs(activeEntities) do
            if isValid(ent) then pcall(ent.remove, ent) end
        end
        print("[Backrooms Engine] All sector entities dematerialized.")
    end)

    -- 3D Rendering Hook: Half-Life 1 Xen Green Zappy Beams
    hook.add("render", "BackroomsRenderXenBeams", function()
        local now = timer.curtime()
        for i = #activeZaps, 1, -1 do
            local zap = activeZaps[i]
            if now > zap.dieTime then
                table.remove(activeZaps, i)
            else
                -- Intense Xen Green Glow
                render.setColor(Color(45, 255, 65, 240))
                if xenBeamMat then
                    render.setMaterial(xenBeamMat)
                end

                local right = zap.ang:getRight()
                local halfW = 90

                -- Render 8 branching, jagged electric lightning arcs
                for b = 1, zap.bolts do
                    local lateral = (b / zap.bolts - 0.5) * (halfW * 1.8)
                    local top = zap.pos + (right * lateral) + Vector(0, 0, 95)
                    local bottom = zap.pos + (right * (lateral + math.random(-25, 25))) - Vector(0, 0, 95)

                    -- Jagged mid-air arcing points that jump every frame
                    local mid1 = top + Vector(math.random(-18, 18), math.random(-18, 18), -55)
                    local mid2 = bottom + Vector(math.random(-18, 18), math.random(-18, 18), 55)

                    local w = math.random(8, 16)
                    render.draw3DBeam(top, mid1, w, 0, 1)
                    render.draw3DBeam(mid1, mid2, w * 0.8, 0, 1)
                    render.draw3DBeam(mid2, bottom, w, 0, 1)
                end
            end
        end
    end)

    -- Reality Shift Pulse
    timer.create("RealityPulseTimer", 150, 0, function()
        pulseAlpha = 220
        pcall(sound.playFile, "weapons/physcannon/energy_sing_explosion2.wav", "2d", function(st)
            if isValid(st) then
                pcall(st.setVolume, st, 0.45)
                pcall(st.play, st)
            end
        end)
    end)

    hook.add("drawhud", "RealityPulseDraw", function()
        if pulseAlpha > 0 then
            pulseColor.a = pulseAlpha
            pcall(render.setColor, pulseColor)
            pcall(render.drawRect, 0, 0, render.getGameResolution())
            pulseAlpha = math.max(0, pulseAlpha - (render.getFrameTime() * 180))
        end
    end)
end
