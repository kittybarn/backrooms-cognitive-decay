-- THE ULTIMATE BACKROOMS PROCEDURAL GENERATOR FOR GARRY'S MOD (STARFALL EX)
-- 
-- Key Highlights in this Test Version:
--  1. Retains 100% Default Textures on Furniture:
--     Desks, chairs, CRT monitors, and electronics spawn with their native,
--     authentic Half-Life 2 wood, plastic, metal, and glass textures.
--  2. Liminal Spatial Density:
--     Reduced clutter. The Backrooms is primarily vast, eerie, empty hallways.
--     Furniture clusters are sparse (8-10% of cells), making encounters memorable.
--  3. Natural Human Offsets & Angles:
--     Chairs are slid out, angled sideways (+/- 25 deg), and offset as though
--     someone hurriedly abandoned them. Monitors are positioned askew on desks.
--  4. Grounded Physics vs. Rare Uncanny Wall Clipping:
--     Normal furniture sits completely flush on the floor (exact Source hull Z math).
--     Rare context-decay anomalies (4% chance) feature desks/chairs partially 
--     fused sideways into walls at dramatic angles (authentic noclip glitches).
--  5. Smooth Drywall Wallpaper & Carpet:
--     Eliminated repeating debris stripes on walls; features smooth matte yellow
--     plaster drywall and damp woven office carpet.
--  6. Burst-Safe Prop Architecture with 100% pcall Protection:
--     Zero crashes from burst limits or invalid engine calls.
-- ==============================================================================

if SERVER then
    -- --------------------------------------------------------------------------
    -- 1. CONFIGURATION & TUNING
    -- --------------------------------------------------------------------------
    local CONFIG = {
        -- Maze Dimensions (Number of grid cells)
        GRID_W = 10,
        GRID_H = 10,
        CELL_SIZE = 189.8,     -- Source units (Matches plate4x4: 4 * 47.45 = 189.8)
        WALL_HEIGHT = 189.8,   -- Solid high partitions, open roof
        
        -- Floor & Wall Models (Standard Hunter modular plates)
        FLOOR_MODEL_LARGE = "models/hunter/plates/plate16x16.mdl", -- 759.2 x 759.2 units per plate
        WALL_MODEL = "models/hunter/plates/plate4x4.mdl",          -- 189.8 x 189.8 units
        
        -- Spawning Throttle (Guarantees zero burst limit overflow)
        SPAWN_INTERVAL = 0.25, -- 4 props per second (matches Starfall token regen)
        
        -- Gameplay Chances (Dialed for authentic liminal Backrooms feel)
        FURNITURE_CHANCE = 0.09,  -- 9% of cells have furniture (rare & eerie)
        GLITCH_CHANCE    = 0.04,  -- 4% chance for rare wall-fused clipping anomaly
        NPC_CHANCE       = 0.07,  -- 7% chance to spawn an uncanny entity in dead-ends
        
        -- Elevation Offset (Safe airspace on gm_functional_flatgrass)
        BASE_ALTITUDE = 0      -- Relative to chip location
    }

    -- --------------------------------------------------------------------------
    -- 2. VERIFIED MATERIALS (Smooth Matte Yellow Drywall & Damp Carpet)
    -- --------------------------------------------------------------------------
    local WALL_PRESETS = {
        { mat = "models/debug/debugwhite",    color = Color(228, 214, 148) }, -- Classic smooth Level 0 yellow wallpaper
        { mat = "phoenix_storms/pack2/panel", color = Color(222, 208, 142) }, -- Liminal panel drywall
        { mat = "models/debug/debugwhite",    color = Color(215, 202, 138) }  -- Darker yellowed hallway drywall
    }

    local FLOOR_PRESETS = {
        { mat = "models/props_c17/furniturefabric001a", color = Color(135, 120, 75) }, -- Damp woven carpet fiber
        { mat = "models/props_c17/furniturefabric002a", color = Color(118, 105, 62) }, -- Musty mildew carpet
        { mat = "models/props_debris/tilefloor001c",    color = Color(175, 170, 150) }  -- Liminal hospital tile
    }

    local PROP_PRESETS = {
        desks = {
            "models/props_c17/FurnitureTable001a.mdl",
            "models/props_c17/FurnitureTable002a.mdl"
        },
        chairs = {
            "models/props_c17/FurnitureChair001a.mdl"
        },
        electronics = {
            "models/props_c17/tv_monitor01.mdl",
            "models/props_lab/monitor01a.mdl"
        },
        storage = {
            "models/props_wasteland/controlroom_filecabinet002a.mdl"
        }
    }

    local NPC_MODELS = {
        "models/humans/group01/male_07.mdl",
        "models/humans/group01/male_02.mdl",
        "models/humans/group02/female_04.mdl",
        "models/kleiner.mdl"
    }

    -- --------------------------------------------------------------------------
    -- 3. BULLETPROOF PCALL PROTECTED HELPERS
    -- --------------------------------------------------------------------------
    local function safeSetMaterial(ent, mat, fallbackColor)
        if not isValid(ent) then return end
        local success = false
        if mat and mat ~= "" then
            local ok, _ = pcall(ent.setMaterial, ent, mat)
            if ok then success = true end
        end
        if not success or fallbackColor then
            if not success then
                pcall(ent.setMaterial, ent, "models/debug/debugwhite")
            end
            if fallbackColor then
                pcall(ent.setColor, ent, fallbackColor)
            end
        end
    end

    local function safeSetSubMaterial(ent, idx, mat)
        if not isValid(ent) or not mat or mat == "" then return end
        pcall(ent.setSubMaterial, ent, idx, mat)
    end

    local function safeSetColor(ent, col)
        if not isValid(ent) or not col then return end
        pcall(ent.setColor, ent, col)
    end

    local function safeSetFlex(ent, idx, weight)
        if not isValid(ent) or not idx then return end
        pcall(ent.setFlexWeight, ent, idx, weight)
    end

    local function safeManipulateBoneAngles(ent, bone, ang)
        if not isValid(ent) or not bone or not ang then return end
        pcall(ent.manipulateBoneAngles, ent, bone, ang)
    end

    -- --------------------------------------------------------------------------
    -- 4. STATE & SPAWN QUEUE
    -- --------------------------------------------------------------------------
    local spawnQueue = {}
    local activeEntities = {}
    local chipOrigin = chip():getPos() + Vector(0, 0, CONFIG.BASE_ALTITUDE)

    -- --------------------------------------------------------------------------
    -- 5. MAZE SOLVER (Recursive Backtracker with Looping Corridors)
    -- --------------------------------------------------------------------------
    local grid = {}
    for x = 1, CONFIG.GRID_W do
        grid[x] = {}
        for y = 1, CONFIG.GRID_H do
            grid[x][y] = {
                visited = false,
                north = true,
                south = true,
                east = true,
                west = true,
                hasFurniture = false,
                isDeadEnd = false
            }
        end
    end

    local function carve(cx, cy)
        grid[cx][cy].visited = true
        local directions = {
            { x = 0, y = 1, wall = "north", opp = "south" },
            { x = 0, y = -1, wall = "south", opp = "north" },
            { x = 1, y = 0, wall = "east", opp = "west" },
            { x = -1, y = 0, wall = "west", opp = "east" }
        }

        -- Shuffle directions
        for i = #directions, 2, -1 do
            local j = math.random(i)
            directions[i], directions[j] = directions[j], directions[i]
        end

        for _, d in ipairs(directions) do
            local nx, ny = cx + d.x, cy + d.y
            if nx >= 1 and nx <= CONFIG.GRID_W and ny >= 1 and ny <= CONFIG.GRID_H then
                if not grid[nx][ny].visited then
                    grid[cx][cy][d.wall] = false
                    grid[nx][ny][d.opp] = false
                    carve(nx, ny)
                elseif math.random() < 0.15 then
                    -- 15% loop chance for authentic Backrooms looping corridors
                    grid[cx][cy][d.wall] = false
                    grid[nx][ny][d.opp] = false
                end
            end
        end
    end

    carve(1, 1)

    -- Identify dead ends
    for x = 1, CONFIG.GRID_W do
        for y = 1, CONFIG.GRID_H do
            local c = grid[x][y]
            local wallCount = (c.north and 1 or 0) + (c.south and 1 or 0) + (c.east and 1 or 0) + (c.west and 1 or 0)
            if wallCount >= 3 then
                c.isDeadEnd = true
            end
        end
    end

    -- --------------------------------------------------------------------------
    -- 6. GREEDY ENQUEUING: FLOORS, WALLS, FURNITURE & NPCS
    -- --------------------------------------------------------------------------
    print("[Backrooms Engine] Synthesizing structural layout...")

    -- A. Large Modular Floor Plates (plate16x16 covers 759x759 units)
    local floorStep = 4 -- Covers 4 cells x 4 cells = 759 units
    local defaultFloor = FLOOR_PRESETS[1]
    for x = 1, CONFIG.GRID_W, floorStep do
        for y = 1, CONFIG.GRID_H, floorStep do
            local fPos = chipOrigin + Vector((x - 1 + floorStep * 0.5) * CONFIG.CELL_SIZE, 
                                            (y - 1 + floorStep * 0.5) * CONFIG.CELL_SIZE, 0)
            table.insert(spawnQueue, {
                pos = fPos,
                ang = Angle(0, 0, 0),
                model = CONFIG.FLOOR_MODEL_LARGE,
                mat = defaultFloor.mat,
                color = defaultFloor.color,
                category = "floor"
            })
        end
    end

    -- B. Wall Partitions (Open Roof, 100% Solid Hunter Plate Collisions)
    for x = 1, CONFIG.GRID_W do
        for y = 1, CONFIG.GRID_H do
            local cell = grid[x][y]
            local cCenter = chipOrigin + Vector((x - 0.5) * CONFIG.CELL_SIZE, (y - 0.5) * CONFIG.CELL_SIZE, 0)
            
            -- Slight wallpaper variation per zone
            local wallPreset = WALL_PRESETS[(x + y) % #WALL_PRESETS + 1]

            -- North Wall
            if cell.north then
                table.insert(spawnQueue, {
                    pos = cCenter + Vector(0, CONFIG.CELL_SIZE * 0.5, CONFIG.WALL_HEIGHT * 0.5),
                    ang = Angle(90, 0, 0),
                    model = CONFIG.WALL_MODEL,
                    mat = wallPreset.mat,
                    color = wallPreset.color,
                    category = "wall"
                })
            end

            -- South Outer Boundary
            if y == 1 and cell.south then
                table.insert(spawnQueue, {
                    pos = cCenter + Vector(0, -CONFIG.CELL_SIZE * 0.5, CONFIG.WALL_HEIGHT * 0.5),
                    ang = Angle(90, 0, 0),
                    model = CONFIG.WALL_MODEL,
                    mat = wallPreset.mat,
                    color = wallPreset.color,
                    category = "wall"
                })
            end

            -- East Wall
            if cell.east then
                table.insert(spawnQueue, {
                    pos = cCenter + Vector(CONFIG.CELL_SIZE * 0.5, 0, CONFIG.WALL_HEIGHT * 0.5),
                    ang = Angle(90, 90, 0),
                    model = CONFIG.WALL_MODEL,
                    mat = wallPreset.mat,
                    color = wallPreset.color,
                    category = "wall"
                })
            end

            -- West Outer Boundary
            if x == 1 and cell.west then
                table.insert(spawnQueue, {
                    pos = cCenter + Vector(-CONFIG.CELL_SIZE * 0.5, 0, CONFIG.WALL_HEIGHT * 0.5),
                    ang = Angle(90, 90, 0),
                    model = CONFIG.WALL_MODEL,
                    mat = wallPreset.mat,
                    color = wallPreset.color,
                    category = "wall"
                })
            end

            -- C. Spatial Affordance Interior Props (Furniture retains default Source materials!)
            if not cell.isDeadEnd and math.random() < CONFIG.FURNITURE_CHANCE then
                local isWallGlitch = math.random() < CONFIG.GLITCH_CHANCE
                
                if isWallGlitch then
                    -- RARE CONTEXT DECAY GLITCH: Table/Chair fused diagonally into the wall (noclip glitch)
                    local glitchPos = cCenter + Vector(math.random(-40, 40), math.random(-40, 40), 18.2)
                    -- Dramatically angled clip (Pitch 25 deg, Roll -18 deg)
                    local glitchAng = Angle(math.random(-25, 25), math.random(0, 360), math.random(-20, 20))
                    
                    table.insert(spawnQueue, {
                        pos = glitchPos,
                        ang = glitchAng,
                        model = PROP_PRESETS.desks[math.random(#PROP_PRESETS.desks)],
                        category = "furniture"
                    })

                    -- Accompanying tilted chair fused nearby
                    table.insert(spawnQueue, {
                        pos = glitchPos + glitchAng:getForward() * -30 + Vector(0, 0, 10),
                        ang = glitchAng + Angle(15, -35, 10),
                        model = PROP_PRESETS.chairs[1],
                        category = "furniture"
                    })
                else
                    -- NORMAL GROUNDED FURNITURE (Exact Source Hull Math - Never Floats)
                    -- Base desk model has hull min Z = -18.2, so Z = 18.2 places feet flush on floor!
                    local deskPos = cCenter + Vector(math.random(-35, 35), math.random(-35, 35), 18.2)
                    local deskYaw = math.random(0, 3) * 90 + math.random(-8, 8) -- Slight human imperfection
                    local deskAng = Angle(0, deskYaw, 0)

                    -- 1. Spawn Desk (Retains native Half-Life 2 wood texture!)
                    table.insert(spawnQueue, {
                        pos = deskPos,
                        ang = deskAng,
                        model = PROP_PRESETS.desks[math.random(#PROP_PRESETS.desks)],
                        category = "furniture"
                    })

                    -- 2. Spawn Monitor perched realistically ON TOP of the desk
                    -- Desk surface is at Z = 37.0; monitor hull min is -8.3; total Z = 45.3
                    local monitorPos = deskPos + Vector(math.random(-8, 8), math.random(-4, 4), 27.1)
                    local monitorAng = Angle(0, deskYaw + math.random(-18, 18), 0) -- Angled slightly askew
                    
                    table.insert(spawnQueue, {
                        pos = monitorPos,
                        ang = monitorAng,
                        model = PROP_PRESETS.electronics[1],
                        category = "furniture"
                    })

                    -- 3. Spawn Chair naturally pulled out and angled
                    -- Chair hull min is -20.1, so Z = 20.1 puts legs flush on floor
                    local chairDistance = math.random(-45, -38)
                    local chairLateralJitter = deskAng:getRight() * math.random(-10, 10)
                    local chairPos = deskPos + (deskAng:getForward() * chairDistance) + chairLateralJitter
                    chairPos.z = chipOrigin.z + 20.1 -- Perfectly grounded!

                    -- Human angle: rotated slightly sideways (+/- 25 deg) as if recently abandoned
                    local chairYaw = deskYaw + math.random(-25, 25)
                    local chairAng = Angle(0, chairYaw, 0)

                    table.insert(spawnQueue, {
                        pos = chairPos,
                        ang = chairAng,
                        model = PROP_PRESETS.chairs[1],
                        category = "furniture"
                    })
                end
            end

            -- D. Rare NPC Entities ("Still Life" & "Stalkers")
            if cell.isDeadEnd and math.random() < CONFIG.NPC_CHANCE then
                local isStillLife = math.random() < 0.60
                local npcPos = cCenter + Vector(0, 0, 10)
                local npcAng = Angle(0, math.random(0, 360), 0)

                table.insert(spawnQueue, {
                    pos = npcPos,
                    ang = npcAng,
                    model = NPC_MODELS[math.random(#NPC_MODELS)],
                    category = "npc",
                    onSpawn = function(ent)
                        -- Freeze Still Life entities
                        pcall(ent.setFrozen, ent, true)

                        -- 1. Facial Flex Deformations (Wide staring eyes, uncanny jaw drop)
                        local okNum, flexCount = pcall(ent.getFlexNum, ent)
                        if okNum and type(flexCount) == "number" then
                            for f = 0, math.min(flexCount - 1, 16) do
                                if math.random() < 0.4 then
                                    safeSetFlex(ent, f, math.random() * 1.5)
                                end
                            end
                        end

                        -- 2. Bone Skeletons Glitch (Twisted neck, elongated spine)
                        local okBone1, neckBone = pcall(ent.lookupBone, ent, "ValveBiped.Bip01_Head1")
                        if okBone1 and neckBone then
                            safeManipulateBoneAngles(ent, neckBone, Angle(math.random(-25, 25), math.random(30, 75), 0))
                        end
                        local okBone2, spineBone = pcall(ent.lookupBone, ent, "ValveBiped.Bip01_Spine2")
                        if okBone2 and spineBone then
                            safeManipulateBoneAngles(ent, spineBone, Angle(math.random(-15, 15), 0, math.random(-20, 20)))
                        end

                        -- 3. SubMaterial Overrides (Uncanny Pitch Black or Wallpaper Face)
                        if math.random() < 0.5 then
                            -- Static TV face / pitch black void
                            safeSetSubMaterial(ent, 0, "models/debug/debugwhite")
                            safeSetColor(ent, Color(0, 0, 0))
                        else
                            -- Wallpaper clothes
                            safeSetSubMaterial(ent, 1, "phoenix_storms/pack2/panel")
                        end

                        print("[Backrooms Engine] ⚠️ Entity Manifested: " .. (isStillLife and "'Still Life'" or "'Stalker'") .. " at " .. tostring(npcPos))
                    end
                })
            end
        end
    end

    local totalQueued = #spawnQueue
    print("[Backrooms Engine] Queue compiled: " .. totalQueued .. " total components.")
    print("[Backrooms Engine] Rate-regulated spawn initiated (0 burst risk)...")

    -- --------------------------------------------------------------------------
    -- 7. ASYNCHRONOUS BURST-REGULATED FIFO SPAWNER WITH PCALL PROTECTION
    -- --------------------------------------------------------------------------
    local spawnIndex = 1
    local spawnedCount = 0

    timer.create("BackroomsSafeSpawner", CONFIG.SPAWN_INTERVAL, 0, function()
        if spawnIndex > #spawnQueue then
            timer.remove("BackroomsSafeSpawner")
            print("[Backrooms Engine] ✅ SECTOR FULLY MATERIALIZED! Active entities: " .. spawnedCount)
            return
        end

        -- Check StarfallEx Burst Quota before EVERY spawn
        if prop.canSpawn() then
            local item = spawnQueue[spawnIndex]
            
            -- Protected spawn call: guaranteed never to throw an unhandled error
            local ok, p = pcall(prop.create, item.pos, item.ang, item.model, true)
            if ok and isValid(p) then
                -- CRITICAL: Only walls and floors have materials applied!
                -- Furniture props RETAIN their authentic native Half-Life 2 materials!
                if item.category == "wall" or item.category == "floor" then
                    safeSetMaterial(p, item.mat, item.color)
                end

                -- Safely execute custom entity callbacks with pcall
                if item.onSpawn then
                    pcall(item.onSpawn, p)
                end
                
                table.insert(activeEntities, p)
                spawnedCount = spawnedCount + 1
                spawnIndex = spawnIndex + 1

                if spawnedCount % 20 == 0 or spawnedCount == totalQueued then
                    local pct = math.floor((spawnedCount / totalQueued) * 100)
                    print("[Backrooms Engine] Progress: " .. spawnedCount .. "/" .. totalQueued .. " (" .. pct .. "%)")
                end
            else
                -- If spawn failed due to engine collision or boundary, skip safely to next
                spawnIndex = spawnIndex + 1
            end
        end
    end)

    -- --------------------------------------------------------------------------
    -- 8. CLEANUP ON REMOVAL
    -- --------------------------------------------------------------------------
    hook.add("Removed", "BackroomsCleanup", function()
        timer.remove("BackroomsSafeSpawner")
        for _, ent in ipairs(activeEntities) do
            if isValid(ent) then 
                pcall(ent.remove, ent)
            end
        end
        print("[Backrooms Engine] All sector entities dematerialized.")
    end)

else
    -- ==========================================================================
    -- CLIENT: AUDIO & VISUAL REALITY PULSE
    -- ==========================================================================
    local pulseAlpha = 0
    local pulseColor = Color(20, 255, 60, 0)

    -- Ambient Zero-Net-Overhead Sound
    hook.add("Initialize", "BackroomsClientInit", function()
        pcall(sound.playFile, "ambient/levels/labs/equipment_printer_loop1.wav", "noplay", function(station)
            if isValid(station) then
                pcall(station.setVolume, station, 0.18)
                pcall(station.play, station)
            end
        end)
    end)

    -- Periodic Reality Shift Pulse Effect
    timer.create("RealityPulseTimer", 120, 0, function()
        pulseAlpha = 220
        pcall(sound.playFile, "ambient/levels/citadel/portal_beam_shoot1.wav", "2d", function(st)
            if isValid(st) then
                pcall(st.setVolume, st, 0.4)
                pcall(st.play, st)
            end
        end)
    end)

    hook.add("drawhud", "RealityPulseDraw", function()
        if pulseAlpha > 0 then
            pulseColor.a = pulseAlpha
            pcall(render.setColor, pulseColor)
            pcall(render.drawRect, 0, 0, render.getGameResolution())
            pulseAlpha = math.max(0, pulseAlpha - (render.getFrameTime() * 180))
        end
    end)
end
