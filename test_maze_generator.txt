--@name Backrooms Test Maze Generator (Burst-Safe Queue, Open Roof)
--@author Cognitive Decay Project
--@server

-- Solves: "The props burst limit has been exceeded"
-- StarfallEx limits prop.create() to ~4 props per second (plyPropBurst).
-- This script uses an asynchronous FIFO spawn queue that checks prop.canSpawn()
-- before materializing each wall and floor, guaranteeing ZERO burst limit errors!

local origin = chip():getPos()
local cellSize = 120 -- Source units per corridor cell
local wallHeight = 110 -- Classic tall Backrooms wall height

-- Maze Grid Settings (5x5 test maze to spawn quickly and safely)
local gridW = 5
local gridH = 5

local spawnQueue = {}
local spawnedProps = {}

print("[Maze Generator] Initializing Burst-Safe Backrooms Maze at " .. tostring(origin))

-- 1. Generate Maze Grid Data (Depth-First Recursive Backtracker)
local grid = {}
for x = 1, gridW do
    grid[x] = {}
    for y = 1, gridH do
        grid[x][y] = { visited = false, north = true, south = true, east = true, west = true }
    end
end

local function carveMaze(cx, cy)
    grid[cx][cy].visited = true
    local directions = {
        { dx = 0, dy = 1, wall = "north", opp = "south" },
        { dx = 0, dy = -1, wall = "south", opp = "north" },
        { dx = 1, dy = 0, wall = "east", opp = "west" },
        { dx = -1, dy = 0, wall = "west", opp = "east" }
    }
    for i = #directions, 2, -1 do
        local j = math.random(1, i)
        directions[i], directions[j] = directions[j], directions[i]
    end

    for _, dir in ipairs(directions) do
        local nx, ny = cx + dir.dx, cy + dir.dy
        if nx >= 1 and nx <= gridW and ny >= 1 and ny <= gridH and not grid[nx][ny].visited then
            grid[cx][cy][dir.wall] = false
            grid[nx][ny][dir.opp] = false
            carveMaze(nx, ny)
        end
    end
end

carveMaze(1, 1)

-- 2. Build Queues for Floors & Walls
local floorMat = "materials/carpet/office_carpet_damp_01"
local wallMat  = "materials/plaster/office_wallpaper_yellow"
local wallModel = "models/hunter/plates/plate1x2.mdl"

-- Enqueue Floors (Using larger plate4x4 / plate2x2 to drastically reduce prop count)
for x = 1, gridW, 2 do
    for y = 1, gridH, 2 do
        local floorPos = origin + Vector((x - 1) * cellSize + cellSize, (y - 1) * cellSize + cellSize, 0)
        table.insert(spawnQueue, {
            pos = floorPos,
            ang = Angle(0, 0, 0),
            model = "models/hunter/plates/plate2x2.mdl",
            mat = floorMat
        })
    end
end

-- Enqueue Walls (Open Roof - No ceiling plates!)
for x = 1, gridW do
    for y = 1, gridH do
        local cellCenter = origin + Vector((x - 1) * cellSize, (y - 1) * cellSize, 0)
        local cellData = grid[x][y]

        if cellData.north then
            table.insert(spawnQueue, {
                pos = cellCenter + Vector(0, cellSize * 0.5, wallHeight * 0.5),
                ang = Angle(90, 0, 0),
                model = wallModel,
                mat = wallMat
            })
        end

        if y == 1 and cellData.south then
            table.insert(spawnQueue, {
                pos = cellCenter + Vector(0, -cellSize * 0.5, wallHeight * 0.5),
                ang = Angle(90, 0, 0),
                model = wallModel,
                mat = wallMat
            })
        end

        if cellData.east then
            table.insert(spawnQueue, {
                pos = cellCenter + Vector(cellSize * 0.5, 0, wallHeight * 0.5),
                ang = Angle(90, 90, 0),
                model = wallModel,
                mat = wallMat
            })
        end

        if x == 1 and cellData.west then
            table.insert(spawnQueue, {
                pos = cellCenter + Vector(-cellSize * 0.5, 0, wallHeight * 0.5),
                ang = Angle(90, 90, 0),
                model = wallModel,
                mat = wallMat
            })
        end
    end
end

local totalQueued = #spawnQueue
print("[Maze Generator] " .. totalQueued .. " components queued. Starting rate-regulated spawn...")

-- 3. Asynchronous Burst-Regulated Spawner
-- Runs on a safe timer tick (every 0.22s), strictly checking prop.canSpawn()
local spawnIndex = 1

timer.create("SafePropSpawner", 0.22, 0, function()
    if spawnIndex > #spawnQueue then
        timer.remove("SafePropSpawner")
        print("[Maze Generator] ✅ MAZE COMPLETELY MATERIALIZED! Total props: " .. #spawnedProps)
        return
    end

    -- Check StarfallEx prop burst quota
    if prop.canSpawn() then
        local item = spawnQueue[spawnIndex]
        local p = prop.create(item.pos, item.ang, item.model, true)
        if isValid(p) then
            p:setMaterial(item.mat)
            table.insert(spawnedProps, p)
            spawnIndex = spawnIndex + 1
        end
    end
end)

-- Clean-up on chip remove
hook.add("Removed", "ClearMazeProps", function()
    timer.remove("SafePropSpawner")
    for _, p in ipairs(spawnedProps) do
        if isValid(p) then p:remove() end
    end
end)
