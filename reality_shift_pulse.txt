--@name Reality Shift (Electric Update Pulse)
--@author Cognitive Decay Project
--@shared

-- Implements rare Backrooms "Zone Reconfiguration / Reality Update":
-- 1. Sudden ambient flicker & buzzing sub-bass
-- 2. Violent electric green / cyan sprite flash & beam discharge (sprites/physbeam + light_glow02)
-- 3. Room geometry instantly mutates / rearranges while the player is blinded

if CLIENT then
    local flashAlpha = 0
    local flashColor = Color(20, 255, 120) -- Electric lime green / radioactive cyan
    local beamActive = false
    local beamOrigin = chip():getPos()
    local beamTargets = {}
    local glowMat = Material("sprites/light_glow02")
    local beamMat = Material("sprites/physbeam")

    -- Screen-space flash and HUD distortion hook
    hook.add("drawhud", "RealityShift_ScreenFlash", function()
        if flashAlpha > 1 then
            render.setColor(Color(flashColor.r, flashColor.g, flashColor.b, math.min(flashAlpha, 255)))
            render.drawRect(0, 0, render.getResolution())
            flashAlpha = math.max(0, flashAlpha - (render.getFrameTime() * 320))
        end
    end)

    -- 3D Electric Arcs & Sprite Corona hook
    hook.add("render", "RealityShift_3DBeams", function()
        if flashAlpha > 10 then
            -- Render 3D glowing sprite ball at room center
            render.setMaterial(glowMat)
            render.drawSprite(beamOrigin + Vector(0, 0, 70), flashAlpha * 1.5, flashAlpha * 1.5, flashColor)

            -- Render crackling electrical beams to corners
            render.setMaterial(beamMat)
            for _, target in ipairs(beamTargets) do
                local jitter = Vector(math.random(-8, 8), math.random(-8, 8), math.random(-8, 8))
                render.drawBeam(beamOrigin + Vector(0,0,70), target + jitter, 16, 0, 1, flashColor)
            end
        end
    end)

    -- Receive update trigger from server
    net.receive("RealityUpdatePulse", function()
        beamOrigin = net.readVector()
        flashAlpha = 255
        beamTargets = {}
        
        -- Generate 6-8 erratic electrical arc targets across the room
        for i = 1, 8 do
            table.insert(beamTargets, beamOrigin + Vector(math.random(-180, 180), math.random(-180, 180), math.random(10, 130)))
        end

        -- Play sharp electrical discharge and ozone pop
        sound.play("ambient/energy/zap" .. math.random(1, 3) .. ".wav", beamOrigin, 85, math.random(90, 115))
        sound.play("ambient/machines/thumper_hit.wav", beamOrigin, 80, 150)
    end)
end

if SERVER then
    util.addNetworkString("RealityUpdatePulse")

    -- Trigger reality shifts at semi-rare intervals (e.g. every 20-45 seconds)
    local function triggerRealityShift()
        local pos = chip():getPos()
        
        net.start("RealityUpdatePulse")
        net.writeVector(pos)
        net.send()

        -- While player is blinded by the electric green flash, rearrange props and walls!
        timer.simple(0.08, function()
            hook.run("OnRealityRestructure", pos)
            print("[Reality Engine] ZONE MUTATION COMPLETE: Reality re-rendered.")
        end)
    end

    -- Random interval timer for unexpected shift
    local function scheduleNextShift()
        local nextDelay = math.random(18, 40)
        timer.simple(nextDelay, function()
            triggerRealityShift()
            scheduleNextShift()
        end)
    end

    scheduleNextShift()
end
