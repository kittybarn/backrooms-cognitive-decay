--@name Backrooms Infinite Mesh & Dynamic Box Clip Maze
--@author Cognitive Decay Project
--@shared

-- ARCHITECTURE TO BYPASS ALL ENGINE LIMITS:
-- 1. VISUALS (Client): Drawn via client-side 3D Meshes (mesh.createFromTable / render.draw3DBox)
--    - Zero props spawned!
--    - Unlimited scale (can cover the entire 32,768 x 32,768 map).
--    - Open roof with classic Level 0 yellow wallpaper.
--
-- 2. SOLID COLLISION (Server): The "Moving Local Clip Box" (Zero Prop Burst)
--    - Instead of spawning 5,000 separate props across the whole map,
--      we spawn only 4 local invisible clip props that follow around the player,
--      repositioning themselves to match whatever walls surround the player's current grid cell!
--    - Exactly 4 props total on the entire server = ZERO burst limit, 100% solid collision!

local mapCellSize = 150 -- Grid size in Source units
local wallHeight = 110

if SERVER then
    -- Spawn exactly 4 local collision clip walls
    -- These 4 props will dynamically reposition around the player to block them from walking through walls!
    local clipModel = "models/hunter/plates/plate1x2.mdl"
    local clipWalls = {}

    local function initLocalCollision()
        for i = 1, 4 do
            local p = prop.create(chip():getPos() + Vector(0, 0, -2000), Angle(0,0,0), clipModel, true)
            if isValid(p) then
                p:setColor(Color(0,0,0,0)) -- Completely invisible
                p:setDrawShadow(false)
                table.insert(clipWalls, p)
            end
        end
        print("[Collision Engine] Dynamic Local Clip Box initialized (4 props total)!")
    end

    timer.simple(0.5, initLocalCollision)

    -- Dynamic Collision Positioning Loop
    -- As the player moves through the maze, update the 4 clip walls to match the player's current cell
    hook.add("think", "UpdateDynamicClipBox", function()
        if #clipWalls < 4 then return end
        local ply = player()
        if not isValid(ply) then return end

        local pPos = ply:getPos()
        local cellX = math.floor(pPos.x / mapCellSize)
        local cellY = math.floor(pPos.y / mapCellSize)
        local cellCenter = Vector((cellX + 0.5) * mapCellSize, (cellY + 0.5) * mapCellSize, pPos.z)

        -- Deterministic maze wall hash (Same logic as client mesh)
        local function hasWall(cx, cy, side)
            local hash = math.sin(cx * 12.9898 + cy * 78.233 + side * 37.719) * 43758.5453
            return (hash - math.floor(hash)) > 0.45
        end

        local offsets = {
            { pos = Vector(0, mapCellSize * 0.5, wallHeight * 0.5), ang = Angle(90, 0, 0), side = 1 },   -- North
            { pos = Vector(0, -mapCellSize * 0.5, wallHeight * 0.5), ang = Angle(90, 0, 0), side = 2 },  -- South
            { pos = Vector(mapCellSize * 0.5, 0, wallHeight * 0.5), ang = Angle(90, 90, 0), side = 3 },  -- East
            { pos = Vector(-mapCellSize * 0.5, 0, wallHeight * 0.5), ang = Angle(90, 90, 0), side = 4 }  -- West
        }

        for i, off in ipairs(offsets) do
            local wallProp = clipWalls[i]
            if isValid(wallProp) then
                if hasWall(cellX, cellY, off.side) then
                    wallProp:setPos(cellCenter + off.pos)
                    wallProp:setAngles(off.ang)
                else
                    -- Move out of the way if no wall
                    wallProp:setPos(cellCenter + Vector(0, 0, -5000))
                end
            end
        end
    end)

    hook.add("Removed", "ClearClips", function()
        for _, p in ipairs(clipWalls) do if isValid(p) then p:remove() end end
    end)
end

if CLIENT then
    local wallMat = Material("materials/plaster/office_wallpaper_yellow")
    local floorMat = Material("materials/carpet/office_carpet_damp_01")

    -- Deterministic wall check (identical to server)
    local function hasWall(cx, cy, side)
        local hash = math.sin(cx * 12.9898 + cy * 78.233 + side * 37.719) * 43758.5453
        return (hash - math.floor(hash)) > 0.45
    end

    -- Client 3D Render Loop: Draws visible corridors around the player
    -- Zero props! Zero server entities! Endless render distance!
    hook.add("render", "DrawInfiniteBackroomsMesh", function()
        local ply = LocalPlayer()
        if not isValid(ply) then return end

        local pPos = ply:getPos()
        local currentCellX = math.floor(pPos.x / mapCellSize)
        local currentCellY = math.floor(pPos.y / mapCellSize)

        local floorZ = pPos.z - 2

        -- Render a radius of 12x12 cells around the player
        local renderRadius = 8

        for x = currentCellX - renderRadius, currentCellX + renderRadius do
            for y = currentCellY - renderRadius, currentCellY + renderRadius do
                local cellCenter = Vector((x + 0.5) * mapCellSize, (y + 0.5) * mapCellSize, floorZ)

                -- 1. DRAW FLOOR (Open Roof above)
                render.setMaterial(floorMat)
                render.draw3DQuad(
                    cellCenter + Vector(-mapCellSize*0.5, -mapCellSize*0.5, 0),
                    cellCenter + Vector(-mapCellSize*0.5,  mapCellSize*0.5, 0),
                    cellCenter + Vector( mapCellSize*0.5,  mapCellSize*0.5, 0),
                    cellCenter + Vector( mapCellSize*0.5, -mapCellSize*0.5, 0)
                )

                -- 2. DRAW WALLS
                render.setMaterial(wallMat)

                -- North Wall
                if hasWall(x, y, 1) then
                    local wPos = cellCenter + Vector(0, mapCellSize * 0.5, 0)
                    render.draw3DQuad(
                        wPos + Vector(-mapCellSize*0.5, 0, 0),
                        wPos + Vector(-mapCellSize*0.5, 0, wallHeight),
                        wPos + Vector( mapCellSize*0.5, 0, wallHeight),
                        wPos + Vector( mapCellSize*0.5, 0, 0)
                    )
                end

                -- East Wall
                if hasWall(x, y, 3) then
                    local wPos = cellCenter + Vector(mapCellSize * 0.5, 0, 0)
                    render.draw3DQuad(
                        wPos + Vector(0, -mapCellSize*0.5, 0),
                        wPos + Vector(0, -mapCellSize*0.5, wallHeight),
                        wPos + Vector(0,  mapCellSize*0.5, wallHeight),
                        wPos + Vector(0,  mapCellSize*0.5, 0)
                    )
                end
            end
        end
    end)
end
